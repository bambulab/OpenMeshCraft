// p4 is query point, p1-p2-p3 is the plane.
// Assume that d1 is positive.

// we first calculate determinant(p1, p2, p3, p4)
// = | p1 1 |
//   | p2 1 |
//   | p3 1 |
//   | p4 1 |
// = | p1 - p4 |
//   | p2 - p4 |
//   | p3 - p4 |
// expand to (we assume that d1 is positive)
// = | d1 * (b1 - p4) + l1 |
//   |       p2 - p4       |
//   |       p3 - p4       |
// and finally reverse the sign of the determinant.

// -- (b1 - p4)
b1p4x = b1x - p4x
b1p4y = b1y - p4y
b1p4z = b1z - p4z

// -- d1 * (b1 - p4)
d1_b1p4x = d1 * b1p4x
d1_b1p4y = d1 * b1p4y
d1_b1p4z = d1 * b1p4z

// -- d1 * (b1 - p4) + l1
p1x_p4x = d1_b1p4x + l1x
p1y_p4y = d1_b1p4y + l1y
p1z_p4z = d1_b1p4z + l1z

// -- (p2 - p4)
p2x_p4x = p2x - p4x
p2y_p4y = p2y - p4y
p2z_p4z = p2z - p4z

// -- (p3 - p4)
p3x_p4x = p3x - p4x
p3y_p4y = p3y - p4y
p3z_p4z = p3z - p4z

// -- m = (d1 * (b1 - p4) + l1) cross (p2 - p4)
t0 = p1x_p4x * p2y_p4y
t1 = p1y_p4y * p2x_p4x
m01 = t0 - t1
t2 = p1x_p4x * p2z_p4z
t3 = p1z_p4z * p2x_p4x
m02 = t2 - t3
t4 = p1y_p4y * p2z_p4z
t5 = p1z_p4z * p2y_p4y
m12 = t4 - t5

// -- m dot (p3 - p4)
mt1 = m01 * p3z_p4z
mt2 = m02 * p3y_p4y
mt3 = m12 * p3x_p4x

// -- reverse sign because we actually calculate a reversed determinant.
mtt = mt2 - mt1
m012 = mtt - mt3
