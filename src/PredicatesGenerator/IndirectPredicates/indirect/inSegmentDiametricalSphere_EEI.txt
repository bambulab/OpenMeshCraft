// pa and pb are the two points of the segment.
// check if pc is inside the smallest (diametral) sphere of the segment.

// it is equivalent to check if the triangle formed by pa, pb and pc is an obutuse triangle.
// sign = (pc - pa) dot (pb - pc)
//      = (lc - dc * pa) dot (pb * dc - lc)
// -- sign > 0 : inside
// -- sign = 0 : on
// -- sign < 0 : outside

// -- dc * pa
dcpax = dc * pax
dcpay = dc * pay
dcpaz = dc * paz

// -- lc - dc * pa
acx = lcx - dcpax
acy = lcy - dcpay
acz = lcz - dcpaz

// -- dc * pb
dcpbx = dc * pbx
dcpby = dc * pby
dcpbz = dc * pbz

// -- dc * pb - lc
cbx = dcpbx - lcx
cby = dcpby - lcy
cbz = dcpbz - lcz

// -- sign
ac_cb_x = acx * cbx
ac_cb_y = acy * cby
ac_cb_z = acz * cbz

sign_xy = ac_cb_x + ac_cb_y
sign = sign_xy + ac_cb_z
