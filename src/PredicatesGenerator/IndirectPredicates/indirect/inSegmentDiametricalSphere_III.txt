// pa and pb are the two points of the segment.
// check if pc is inside the smallest (diametral) sphere of the segment.

// it is equivalent to check if the triangle formed by pa, pb and pc is an obutuse triangle.
// sign = (pc - pa) dot (pb - pc)
//      = (da * lc - dc * la) dot (dc * lb - db * lc)
// (we ensure that da, db and dc are positive)
// -- sign > 0 : inside
// -- sign = 0 : on
// -- sign < 0 : outside

// -- da * lc
dalcx = da * lcx
dalcy = da * lcy
dalcz = da * lcz

// -- dc * la
dclax = dc * lax
dclay = dc * lay
dclaz = dc * laz

// -- da * lc - dc * la
acx = dalcx - dclax
acy = dalcy - dclay
acz = dalcz - dclaz

// -- dc * lb
dclbx = dc * lbx
dclby = dc * lby
dclbz = dc * lbz

// -- db * lc
dblcx = db * lcx
dblcy = db * lcy
dblcz = db * lcz

// -- dc * lb - db * lc
cbx = dclbx - dblcx
cby = dclby - dblcy
cbz = dclbz - dblcz

// -- sign = (pc - pa) dot (pb - pc)
ac_cb_x = acx * cbx
ac_cb_y = acy * cby
ac_cb_z = acz * cbz

sign_xy = ac_cb_x + ac_cb_y
sign = sign_xy + ac_cb_z
