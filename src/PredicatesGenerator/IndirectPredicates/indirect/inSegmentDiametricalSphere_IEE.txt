// pa and pb are the two points of the segment.
// check if pc is inside the smallest (diametral) sphere of the segment.

// it is equivalent to check if the triangle formed by pa, pb and pc is an obutuse triangle.
// sign = (pb - pa).squaredNorm() - ((pc - pa).squaredNorm() + (pc - pb).squaredNorm())
//      = (da * pb - la).squaredNorm() - ((da * pc - la).squaredNorm() + da^2 * (pc - pb).squaredNorm())
// -- sign > 0 : inside
// -- sign = 0 : on
// -- sign < 0 : outside

// -- da * pb
dapbx = da * pbx
dapby = da * pby
dapbz = da * pbz

// -- da * pb - la
abx = dapbx - lax
aby = dapby - lay
abz = dapbz - laz

// -- (da * pb - la).squaredNorm
abx2 = abx * abx
aby2 = aby * aby
abz2 = abz * abz
t0 = abx2 + aby2
absqr = t0 + abz2

// -- da * pc
dapcx = da * pcx
dapcy = da * pcy
dapcz = da * pcz

// -- da * pc - la
acx = dapcx - lax
acy = dapcy - lay
acz = dapcz - laz

// -- (da * pc - la).squaredNorm
acx2 = acx * acx
acy2 = acy * acy
acz2 = acz * acz
t1 = acx2 + acy2
acsqr = t1 + acz2

// -- pc - pb
bcx = pcx - pbx
bcy = pcy - pby
bcz = pcz - pbz

// -- (pc - pb).squaredNorm
bcx2 = bcx * bcx
bcy2 = bcy * bcy
bcz2 = bcz * bcz
t2 = bcx2 + bcy2
bcsqr = t2 + bcz2

// -- da^2 * (pc - pb).squaredNorm
da2 = da * da
da2bcsqr = da2 * bcsqr

// -- sign
ac_bc = acsqr + da2bcsqr
d = absqr - ac_bc
