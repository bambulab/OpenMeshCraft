// pa and pb are the two points of the segment.
// check if pc is inside the smallest (diametral) sphere of the segment.

// it is equivalent to check if the triangle formed by pa, pb and pc is an obutuse triangle.
// sign = (pc - pa) dot (pb - pc)
//      = (da * pc - la) dot (pb - pc)
// (we ensure that da is always positive.)
// -- sign > 0 : inside
// -- sign = 0 : on
// -- sign < 0 : outside

// -- da * pc
dapcx = da * pcx
dapcy = da * pcy
dapcz = da * pcz

// -- da * pc - la
acx = dapcx - lax
acy = dapcy - lay
acz = dapcz - laz

// -- pb - pc
cbx = pbx - pcx
cby = pby - pcy
cbz = pbz - pcz

// -- sign
ac_cb_x = acx * cbx
ac_cb_y = acy * cby
ac_cb_z = acz * cbz

sign_xy = ac_cb_x + ac_cb_y
sign = sign_xy + ac_cb_z
