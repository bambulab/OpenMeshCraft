implicitPoint2D(r:lrx,lry,dr)
implicitPoint2D(q:lqx,lqy,dq)

// p, r and q form two vectors (p-q) and (r-q), q is the common point.
// dotProductSign2D: (p-q) dot (r-q).
// Assume that dp > 0, dr > 0 and dq > 0.

//   (p-q) dot (r-q)
// = (lpx / dp - lqx / dq)(lrx / dr - lqx / dq) + (lpy / dp - lqy / dq)(lry / dr - lqy / dq)
// = (lpx * dq - lqx * dp)(lrx * dq - lqx * dr) + (lpy * dq - lqy * dp)(lry * dq - lqy * dr)

lqxdp = lqx * dp
lqydp = lqy * dp
lpxdq = lpx * dq
lpydq = lpy * dq
lx = lpxdq - lqxdp
ly = lpydq - lqydp
lqxdr = lqx * dr
lqydr = lqy * dr
lrxdq = lrx * dq
lrydq = lry * dq
gx = lrxdq - lqxdr
gy = lrydq - lqydr
dx = lx * gx
dy = ly * gy
d = dx + dy
