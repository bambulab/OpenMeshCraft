// p4 is the query point, p1, p2 and p3 form the plane.
// Assume that d1 and d2 is positive.

// we first calculate determinant(p1, p2, p3, p4)
// = | p1 1 |
//   | p2 1 |
//   | p3 1 |
//   | p4 1 |
// = | p1 - p4 |
//   | p2 - p4 |
//   | p3 - p4 |
// expand to (we assume that d1 and d2 are positive)
//   | l1 - d1 * p4 |
//   | l2 - d2 * p4 |
//   | p3 - p4      |
// and finally reverse the sign of the determinant.

// -- d1 * p4
d1p4x = d1 * p4x
d1p4y = d1 * p4y
d1p4z = d1 * p4z

// -- d2 * p4
d2p4x = d2 * p4x
d2p4y = d2 * p4y
d2p4z = d2 * p4z

// -- l1 - d1 * p4
p1p4x = l1x - d1p4x
p1p4y = l1y - d1p4y
p1p4z = l1z - d1p4z

// -- l2 - d2 * p4
p2p4x = l2x - d2p4x
p2p4y = l2y - d2p4y
p2p4z = l2z - d2p4z

// -- p3 - p4
p3p4x = p3x - p4x
p3p4y = p3y - p4y
p3p4z = p3z - p4z

// -- m = (p1 - p4) cross (p2 - p4)
t0 = p1p4x * p2p4y
t1 = p1p4y * p2p4x
m01 = t0 - t1
t2 = p1p4x * p2p4z
t3 = p1p4z * p2p4x
m02 = t2 - t3
t4 = p1p4y * p2p4z
t5 = p1p4z * p2p4y
m12 = t4 - t5

// -- m dot (p3 - p4)
mt1 = m01 * p3p4z
mt2 = m02 * p3p4y
mt3 = m12 * p3p4x

// -- reverse sign because we actually calculate a reversed determinant.
mtt = mt2 - mt1
m012 = mtt - mt3
