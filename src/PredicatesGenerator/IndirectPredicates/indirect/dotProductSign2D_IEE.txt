explicitPoint2D(r:rx,ry)
explicitPoint2D(q:qx,qy)

// p, r and q form two vectors (p-q) and (r-q), q is the common point.
// dotProductSign2D: (p-q) dot (r-q).
// Assume that dp > 0.

//   (p-q) dot (r-q)
// = (lpx / dp - qx)(rx - qx) + (lpy / dp - qy)(ry - qy)
// = (lpx - qx * dp)(rx - qx) + (lpy - qy * dp)(ry - qy)

qxdp = qx * dp
qydp = qy * dp
lx = lpx - qxdp
ly = lpy - qydp
gx = rx - qx
gy = ry - qy
dx = lx * gx
dy = ly * gy
d = dx + dy
