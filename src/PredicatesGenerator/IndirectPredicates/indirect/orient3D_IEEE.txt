// p4 is the query point, p1, p2 and p3 form the plane.
// Assume that d1 is positive.

// we first calculate determinant(p1, p2, p3, p4)
// = | p1 1 |
//   | p2 1 |
//   | p3 1 |
//   | p4 1 |
// = | p1 - p4 |
//   | p2 - p4 |
//   | p3 - p4 |
// expand to (we assume that d1 is positive)
// = | l1 - p4 * d1 |
//   | p2 - p4      |
//   | p3 - p4      |
// and finally reverse the sign of the determinant.

// -- p4 * d1
dp4x = d1 * p4x
dp4y = d1 * p4y
dp4z = d1 * p4z

// -- l1 - p4 * d1
ix_p4x = l1x - dp4x
iy_p4y = l1y - dp4y
iz_p4z = l1z - dp4z

// -- p2 - p4
p2x_p4x = p2x - p4x
p2y_p4y = p2y - p4y
p2z_p4z = p2z - p4z

// -- p3 - p4
p3x_p4x = p3x - p4x
p3y_p4y = p3y - p4y
p3z_p4z = p3z - p4z

// -- (l1 - p4 * d1) cross (p2 - p4)
t0 = ix_p4x * p2y_p4y
t1 = iy_p4y * p2x_p4x
m01 = t0 - t1
t2 = ix_p4x * p2z_p4z
t3 = iz_p4z * p2x_p4x
m02 = t2 - t3
t4 = iy_p4y * p2z_p4z
t5 = iz_p4z * p2y_p4y
m12 = t4 - t5

// -- (p3 - p4) dot ((l1 - p4 * d1) cross (p2 - p4))
mt1 = m01 * p3z_p4z
mt2 = m02 * p3y_p4y
mt3 = m12 * p3x_p4x

// -- reverse sign because we actually calculate a reversed determinant.
mtt = mt2 - mt1
m012 = mtt - mt3
