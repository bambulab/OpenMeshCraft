explicitPoint2D(r:rx,ry)
implicitPoint2D(q:lqx,lqy,dq)

// p, r and q form two vectors (p-q) and (r-q), q is the common point.
// dotProductSign2D: (p-q) dot (r-q).
// Assume that dp > 0 and dq > 0.

//   (p-q) dot (r-q)
// = (lpx / dp - lqx / dq)(rx - lqx / dq) + (lpy / dp - lqy / dq)(ry - lqy / dq)
// = (lpx * dq - lqx * dp)(rx * dq - lqx) + (lpy * dq - lqy * dp)(ry * dq - lqy)

lpxdq = lpx * dq
lpydq = lpy * dq
lqxdp = lqx * dp
lqydp = lqy * dp
rxq = rx * dq
ryq = ry * dq
lx = lpxdq - lqxdp
ly = lpydq - lqydp
gx = rxq - lqx
gy = ryq - lqy
dx = lx * gx
dy = ly * gy
d = dx + dy
