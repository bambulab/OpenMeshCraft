implicitPoint2D(r:lrx,lry,dr)
explicitPoint2D(q:qx,qy)

// p, r and q form two vectors (p-q) and (r-q), q is the common point.
// dotProductSign2D: (p-q) dot (r-q).
// Assume that dp > 0 and dr > 0.

//   (p-q) dot (r-q)
// = (lpx / dp - qx)(lrx / dr - qx) + (lpy / dp - qy)(lry / dr - qy)
// = (lpx - qx * dp)(lrx - qx * dr) + (lpy - qy * dp)(lry - qy * dr)

qxdp = qx * dp
qydp = qy * dp
lx = lpx - qxdp
ly = lpy - qydp
qxdr = qx * dr
qydr = qy * dr
gx = lrx - qxdr
gy = lry - qydr
dx = lx * gx
dy = ly * gy
d = dx + dy
