explicitPoint2D(p:px,py)
explicitPoint2D(r:rx,ry)
explicitPoint2D(q:qx,qy)
explicitPoint2D(s:sx,sy)

// p, r, q and s form two vectors (p-q) and (r-s).
// dotProductSign2D: (p-q) dot (r-s).

//   (p-q) dot (r-s)
// = (px - qx)(rx - sx) + (py - qy)(ry - sy)

lx = px - qx
ly = py - qy
gx = rx - sx
gy = ry - sy
dx = lx * gx
dy = ly * gy
d = dx + dy
