explicitPoint3D(p:px,py,pz)
explicitPoint3D(r:rx,ry,rz)
explicitPoint3D(q:qx,qy,qz)

// p, r and q form two vectors (p-q) and (r-q), q is the common point.
// dotProductSign2D: (p-q) dot (r-q).

//   (p-q) dot (r-q)
// = (px - qx)(rx - qx) + (py - qy)(ry - qy) + (pz - qz)(rz - qz)

lx = px - qx
ly = py - qy
lz = pz - qz
gx = rx - qx
gy = ry - qy
gz = rz - qz
dx = lx * gx
dy = ly * gy
dz = lz * gz
d1 = dx + dy
d = d1 + dz
