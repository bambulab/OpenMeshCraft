explicitPoint3D(pa:pax,pay,paz)
explicitPoint3D(pb:pbx,pby,pbz)
explicitPoint3D(pc:pcx,pcy,pcz)

// pa and pb are the two points of the segment.
// check if pc is inside the smallest (diametral) sphere of the segment.

// it is equivalent to check if the triangle formed by pa, pb and pc is an obutuse triangle.
// sign = (pb - pa).squaredNorm() - ((pc - pa).squaredNorm() + (pc - pb).squaredNorm())
// -- sign > 0 : inside
// -- sign = 0 : on
// -- sign < 0 : outside

// -- b - a
abx = pbx - pax
aby = pby - pay
abz = pbz - paz

// -- (b - a).squaredNorm
abx2 = abx * abx
aby2 = aby * aby
abz2 = abz * abz
t0 = abx2 + aby2
absqr = t0 + abz2

// -- c - a
acx = pcx - pax
acy = pcy - pay
acz = pcz - paz

// -- (c - a).squaredNorm
acx2 = acx * acx
acy2 = acy * acy
acz2 = acz * acz
t1 = acx2 + acy2
acsqr = t1 + acz2

// -- c - b
bcx = pcx - pbx
bcy = pcy - pby
bcz = pcz - pbz

// -- (c - b).squaredNorm
bcx2 = bcx * bcx
bcy2 = bcy * bcy
bcz2 = bcz * bcz
t2 = bcx2 + bcy2
bcsqr = t2 + bcz2

// -- sign = (b - a).squaredNorm - ((c - a).squaredNorm + (c - b).squaredNorm)
ac_bc = acsqr + bcsqr
d = absqr - ac_bc
