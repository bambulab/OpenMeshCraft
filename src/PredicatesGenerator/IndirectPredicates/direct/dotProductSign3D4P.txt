explicitPoint3D(p:px,py,pz)
explicitPoint3D(r:rx,ry,rz)
explicitPoint3D(q:qx,qy,qz)
explicitPoint3D(s:sx,sy,sz)

// p, r, q and s form two vectors (p-q) and (r-s).
// dotProductSign2D: (p-q) dot (r-s).

//   (p-q) dot (r-s)
// = (px - qx)(rx - sx) + (py - qy)(ry - sy) + (pz - qz)(rz - sz)

lx = px - qx
ly = py - qy
lz = pz - qz
gx = rx - sx
gy = ry - sy
gz = rz - sz
dx = lx * gx
dy = ly * gy
dz = lz * gz
d1 = dx + dy
d = d1 + dz
